# Install with:
#   pip install -r requirements-dev.txt            # core + tests
#   pip install -r requirements.txt \              # full app incl. GPU
#       --extra-index-url https://download.pytorch.org/whl/cu124
#
# The torch/torchvision pair is deliberately kept on the cu124 wheels: newer
# CUDA builds are ~3 GB to download and rfdetr 1.11 is validated against this
# matrix, so upgrading it is a separate, measurable change.
